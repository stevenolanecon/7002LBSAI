{
  const vals = live_student_means.map(d => d.mean_rt)
  if (vals.length < 2) return html`<p style="color:#aaa;">Not enough data yet.</p>`
  const n = Math.min(boot_controls.n, vals.length)
  const B = boot_controls.B
  const means = []
  for (let b = 0; b < B; b++) {
    let sum = 0
    for (let i = 0; i < n; i++) sum += vals[Math.floor(Math.random() * vals.length)]
    means.push(sum / n)
  }
  const mu = d3.mean(means)
  const se = d3.deviation(means)
  const pad = Math.max(10, (d3.max(means) - d3.min(means)) * 0.15)
  const xmin = d3.min(means) - pad
  const xmax = d3.max(means) + pad

  return Plot.plot({
    height: 450,
    marginLeft: 55,
    marginBottom: 36,
    x: { label: "Bootstrap sample mean (ms)", domain: [xmin, xmax] },
    y: { label: "Count" },
    marks: [
      Plot.rectY(means, Plot.binX({ y: "count" }, { x: d => d, thresholds: 30, fill: "#457b9d", fillOpacity: 0.75 })),
      Plot.ruleX([0]),
      Plot.ruleX([mu_pop], { stroke: "#e63946", strokeWidth: 2.5, strokeDasharray: "6,4",
                                      tip: true, title: () => `Original sample mean: ${mu_pop} ms` }),
      Plot.text([{ x: xmax, y: 0, label: `bootstrap mean = ${mu.toFixed(1)} · bootstrap SE = ${se.toFixed(1)}` }],
                { x: "x", text: "label", frameAnchor: "top-right", dy: 10, fill: "#333", fontSize: 12 }),
    ]
  })
}
