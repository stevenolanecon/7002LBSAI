Pooled histogram — the whole class

A live tool for the workshop

// Screen readers walk into an SVG's own children unless told otherwise — see
// wk1/week1.qmd for the full rationale. Same helper, reused here so this
// standalone chart gets the same one-sentence description treatment.
function accessibleChart(node, label) {
  node.setAttribute("aria-hidden", "true")
  const wrapper = document.createElement("div")
  wrapper.setAttribute("role", "img")
  wrapper.setAttribute("aria-label", label)
  wrapper.appendChild(node)
  return wrapper
}
// Same 1.5×IQR outlier fence the rest of Week 1 uses (see wk1/week1.qmd's
// pool_fence) — the default "exclude above" cutoff, not the true max, so
// the chart opens readable rather than stretched out by a handful of
// extreme trials. Guarded for very sparse data (early in the workshop,
// before enough submissions exist to compute quartiles meaningfully).
pool_fence = {
  const vals = live_data.map(d => d.reaction_time_ms)
  const thrMax = Math.ceil(live_pool_max / 10) * 10 + 10
  if (vals.length < 4) return thrMax
  const sorted = [...vals].sort((a, b) => a - b)
  const q1 = d3.quantile(sorted, 0.25)
  const q3 = d3.quantile(sorted, 0.75)
  const iqr = q3 - q1
  return Math.min(thrMax, Math.ceil((q3 + 1.5 * iqr) / 10) * 10)
}
viewof hist_controls = {
  const thrMax = Math.ceil(live_pool_max / 10) * 10 + 10

  const f = html`<div style="display:flex;align-items:center;gap:24px;flex-wrap:wrap;
                              font-size:13px;font-family:'Helvetica Neue',sans-serif;
                              padding:6px 2px;color:#333;">
    <label style="display:flex;align-items:center;gap:8px;">
      <span style="color:#666;white-space:nowrap;">Bin width</span>
      <input type="range" name="bin_width" min="10" max="80" step="10" value="30"
             style="width:100px;cursor:pointer;accent-color:#457b9d;">
      <span id="bw-label" style="font-weight:600;min-width:34px;color:#457b9d;">30 ms</span>
    </label>
    <label style="display:flex;align-items:center;gap:8px;">
      <span style="color:#e63946;white-space:nowrap;font-weight:600;">Exclude above</span>
      <input type="range" name="threshold" min="${live_pool_min}" max="${thrMax}" step="10" value="${pool_fence}"
             style="width:110px;cursor:pointer;accent-color:#e63946;">
      <span id="thr-label" style="font-weight:600;min-width:40px;color:#e63946;">${pool_fence >= thrMax ? 'All' : pool_fence + ' ms'}</span>
    </label>
    <label style="display:flex;align-items:center;gap:6px;cursor:pointer;">
      <input type="checkbox" name="show_mean" checked style="cursor:pointer;accent-color:#e63946;"> Mean
    </label>
    <label style="display:flex;align-items:center;gap:6px;cursor:pointer;">
      <input type="checkbox" name="show_median" style="cursor:pointer;accent-color:#f4a261;"> Median
    </label>
    <label style="display:flex;align-items:center;gap:6px;cursor:pointer;">
      <input type="checkbox" name="show_kde" style="cursor:pointer;accent-color:#2d6a4f;"> KDE
    </label>
    <label style="display:flex;align-items:center;gap:6px;cursor:pointer;">
      <input type="checkbox" name="show_pct" style="cursor:pointer;accent-color:#333;"> Show as %
    </label>
  </div>`

  const bwRange  = f.querySelector('[name=bin_width]')
  const bwLabel  = f.querySelector('#bw-label')
  const thrRange = f.querySelector('[name=threshold]')
  const thrLabel = f.querySelector('#thr-label')

  bwRange.addEventListener('input', () => {
    bwLabel.textContent = bwRange.value + ' ms'
    f.dispatchEvent(new CustomEvent('input'))
  })
  thrRange.addEventListener('input', () => {
    thrLabel.textContent = +thrRange.value >= thrMax ? 'All' : thrRange.value + ' ms'
    f.dispatchEvent(new CustomEvent('input'))
  })
  f.querySelectorAll('input[type=checkbox]').forEach(cb =>
    cb.addEventListener('change', () => f.dispatchEvent(new CustomEvent('input')))
  )
  Object.defineProperty(f, 'value', { get() { return {
    bin_width:  +f.querySelector('[name=bin_width]').value,
    threshold:  +f.querySelector('[name=threshold]').value,
    show_mean:   f.querySelector('[name=show_mean]').checked,
    show_median: f.querySelector('[name=show_median]').checked,
    show_kde:    f.querySelector('[name=show_kde]').checked,
    show_pct:    f.querySelector('[name=show_pct]').checked,
  }}})
  return f
}
{
  if (live_data.length < 2) return html`<p style="color:#aaa;">Not enough data yet — check back once more of the class has submitted.</p>`

  const thr      = hist_controls.threshold
  const thrMax   = Math.ceil(live_pool_max / 10) * 10 + 10
  const filtered = live_data.filter(d => d.reaction_time_ms <= thr)
  const vals     = filtered.map(d => d.reaction_time_ms)
  const mu       = d3.mean(vals)
  const med      = d3.median(vals)
  const sigma    = d3.deviation(vals)
  const excluded = live_data.length - filtered.length
  const bw       = hist_controls.bin_width
  const pct      = hist_controls.show_pct

  const pad  = Math.max(20, Math.round((d3.max(vals) - d3.min(vals)) * 0.1))
  const xmin = Math.floor((d3.min(vals) - pad) / 10) * 10
  const xmax = Math.ceil((thr + pad) / 10) * 10

  // KDE — Epanechnikov kernel, Silverman's bandwidth
  const h = sigma * 1.06 * Math.pow(vals.length, -0.2)
  const kdePoints = (sigma > 0) ? d3.range(xmin, xmax, 2).map(x => {
    const density = d3.mean(vals, v => {
      const u = (x - v) / h
      return Math.abs(u) <= 1 ? 0.75 * (1 - u * u) / h : 0
    })
    return { x, y: pct ? density * bw : density * filtered.length * bw }
  }) : []

  const chartLabel = `This chart shows how bin width and where you cut off outliers change a histogram's apparent shape. Currently: bins ${bw} ms wide` +
    (excluded > 0 ? `, with ${excluded} value${excluded > 1 ? 's' : ''} above ${thr} ms excluded` : ", with no values excluded") +
    `. This is the class's real, in-progress reaction time data.` +
    (hist_controls.show_mean ? ` The mean is ${mu.toFixed(1)} ms.` : "") +
    (hist_controls.show_median ? ` The median is ${med.toFixed(1)} ms.` : "") +
    (hist_controls.show_kde ? " A smoothed density curve is overlaid on the bars." : "")

  const svg = Plot.plot({
    height: 480,
    marginLeft: 55,
    marginBottom: 32,
    x: { label: "Reaction time (ms)", domain: [xmin, xmax] },
    y: pct
      ? { label: "% of trials", percent: true }
      : { label: "Count" },
    marks: [
      Plot.rectY(filtered,
        Plot.binX({ y: pct ? "proportion" : "count" },
          { x: "reaction_time_ms", thresholds: d3.range(xmin, xmax + bw, bw),
            fill: "#457b9d", fillOpacity: 0.75 })),
      Plot.ruleX([0]),
      hist_controls.show_mean   ? Plot.ruleX([mu],  { stroke: "#e63946", strokeWidth: 2.5,
                             strokeDasharray: "6,4",
                             tip: true, title: () => `Mean: ${mu.toFixed(1)} ms` }) : null,
      hist_controls.show_median ? Plot.ruleX([med], { stroke: "#f4a261", strokeWidth: 2.5,
                             strokeDasharray: "6,4",
                             tip: true, title: () => `Median: ${med.toFixed(1)} ms` }) : null,
      hist_controls.show_kde    ? Plot.line(kdePoints, { x: "x", y: "y",
                             stroke: "#2d6a4f", strokeWidth: 2.5 }) : null,
      excluded > 0 ? Plot.text([{ x: xmax, y: 0 }], {
                             text: () => `${excluded} value${excluded > 1 ? 's' : ''} excluded`,
                             x: "x", frameAnchor: "top-right",
                             fill: "#e63946", fontSize: 12, fontStyle: "italic" }) : null,
    ]
  })

  return accessibleChart(svg, chartLabel)
}
{
  const box = document.createElement('div')
  if (hist_controls.show_kde) {
    box.innerHTML = `<div style="margin-top:10px;background:#eaf5ef;border-left:4px solid #2d6a4f;
                        border-radius:6px;padding:10px 16px;text-align:left;font-size:0.8em;
                        color:#333;line-height:1.5;">
        <strong style="color:#1e3a2f;">What is a KDE?</strong><br>
        A kernel density estimate smooths the histogram into a continuous curve. Instead of
        counting values into fixed-width bins, it places a small "bump" over every observation
        and adds them up — so the shape it shows doesn't depend on a bin-width choice the way
        the bars do.
      </div>`
  }
  return box
}

This uses the real, in-progress class data — try it during the workshop as results come in. Extreme outliers are excluded by default (the same 1.5×IQR rule used in the workshop); drag the “exclude above” threshold up to bring them back, or down further, and see how much the mean moves compared to the median.
