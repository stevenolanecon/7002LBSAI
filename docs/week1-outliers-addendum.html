function accessibleChart(node, label) {
  node.setAttribute("aria-hidden", "true")
  const wrapper = document.createElement("div")
  wrapper.setAttribute("role", "img")
  wrapper.setAttribute("aria-label", label)
  wrapper.appendChild(node)
  return wrapper
}
// Everything on this deck is computed live from the real data — nothing
// here is a hardcoded number from a one-off analysis, so this stays
// accurate if the Sheet is ever refreshed or corrected.
analysis = {
  const vals = data.map(d => d.reaction_time_ms)
  const sorted = [...vals].sort((a, b) => a - b)
  const n = vals.length
  const mean = d3.mean(vals)
  const median = d3.median(vals)
  const sd = d3.deviation(vals)
  const q1 = d3.quantile(sorted, 0.25)
  const q3 = d3.quantile(sorted, 0.75)
  const iqr = q3 - q1
  const fenceHi = q3 + 1.5 * iqr

  const outliers = data.filter(d => d.reaction_time_ms > fenceHi)
  const nOut = outliers.length
  const pctOut = (nOut / n) * 100

  const studentsAll = new Set(data.map(d => d.student_name))
  const studentsOut = new Set(outliers.map(d => d.student_name))

  const byTrial = d3.rollup(outliers, v => v.length, d => d.trial)
  const byDeviceOut = d3.rollup(outliers, v => v.length, d => d.device)
  const byDeviceAll = d3.rollup(data, v => v.length, d => d.device)
  const byCaffOut = d3.rollup(outliers, v => v.length, d => d.caffeine)
  const byCaffAll = d3.rollup(data, v => v.length, d => d.caffeine)

  const perStudentOutCount = d3.rollup(outliers, v => v.length, d => d.student_name)
  const buckets = { zero: 0, oneTwo: 0, threePlus: 0 }
  for (const s of studentsAll) {
    const c = perStudentOutCount.get(s) || 0
    if (c === 0) buckets.zero++
    else if (c <= 2) buckets.oneTwo++
    else buckets.threePlus++
  }

  // Trials 1–2 share of all outliers, vs. the 20% you'd expect if outliers
  // were spread evenly across all 10 trial positions.
  const earlyOutliers = (byTrial.get(1) || 0) + (byTrial.get(2) || 0)
  const earlyPct = nOut ? (earlyOutliers / nOut) * 100 : 0

  return {
    n, mean, median, sd, q1, q3, iqr, fenceHi,
    nOut, pctOut, earlyOutliers, earlyPct,
    studentsAllCount: studentsAll.size, studentsOutCount: studentsOut.size,
    byTrial, byDeviceOut, byDeviceAll, byCaffOut, byCaffAll, buckets
  }
}
// The recommended treatment, applied consistently to whatever the live
// data currently holds. "Error" isn't a fixed millisecond cutoff — a
// slow-but-real distracted trial can genuinely run several seconds long.
// Instead, a flagged trial only counts as a clear error if it's wildly
// separated from every other flagged trial (more than 2.5x the
// next-highest) — the same test that isolated the single ~56-second
// trial excluded live in the workshop. Anything flagged but not that
// extreme is a real value: dropped with a note at trial positions 3–10
// (inattention, not the ability we're measuring), kept at positions 1–2
// (a genuine warm-up effect).
recommendation = {
  const flagged = data.filter(d => d.reaction_time_ms > analysis.fenceHi)
  const bySize = [...flagged].sort((a, b) => b.reaction_time_ms - a.reaction_time_ms)

  const errorRows = []
  if (bySize.length >= 2 && bySize[0].reaction_time_ms > 2.5 * bySize[1].reaction_time_ms) {
    errorRows.push(bySize[0])
  }
  const errorIds = new Set(errorRows.map(d => `${d.student_name}|${d.trial}`))
  const remaining = flagged.filter(d => !errorIds.has(`${d.student_name}|${d.trial}`))
  const softDrop = remaining.filter(d => d.trial > 2)
  const keptEarly = remaining.filter(d => d.trial <= 2)

  const dropIds = new Set([...errorRows, ...softDrop].map(d => `${d.student_name}|${d.trial}`))
  const finalRows = data.filter(d => !dropIds.has(`${d.student_name}|${d.trial}`))

  function stats(rows) {
    const vals = rows.map(d => d.reaction_time_ms)
    const sorted = [...vals].sort((a, b) => a - b)
    return {
      n: vals.length,
      mean: d3.mean(vals),
      sd: d3.deviation(vals),
      min: d3.min(vals),
      max: d3.max(vals),
      median: d3.median(vals),
      iqr: d3.quantile(sorted, 0.75) - d3.quantile(sorted, 0.25),
    }
  }

  const nError = errorRows.length
  const nSoftDrop = softDrop.length
  const nKeptEarly = keptEarly.length

  const errorLine = nError > 0
    ? `**Drop** the trial at ${(errorRows[0].reaction_time_ms / 1000).toFixed(0)} seconds — several times longer than every other flagged trial, not a real single-tap reaction time, just an error.`
    : `**Drop** any trial that's wildly separated from every other flagged trial — none of the current data is that extreme, but the rule still stands.`
  const softDropLine = `**Drop, with a note**, the remaining flagged trials at positions 3–10 (${nSoftDrop} of them) — real values, but driven by inattention rather than the reaction-time ability we're actually measuring`
  const keptEarlyLine = `**Keep** the flagged trials at positions 1–2 (${nKeptEarly} of them) — a genuine warm-up effect, not noise to remove`

  return {
    nError, nSoftDrop, nKeptEarly,
    maxErrorMs: errorRows.length ? errorRows[0].reaction_time_ms : null,
    errorLine, softDropLine, keptEarlyLine,
    raw: stats(data),
    final: stats(finalRows),
  }
}

Addendum: what our real data actually showed

In the workshop, we excluded one obvious outlier. That wasn’t the whole story.

Quick recap

In the Week 1 workshop, we spotted one clearly broken trial — one value at around 56 seconds, obviously not a real reaction time — and excluded it live so the histogram and box plot were readable.

If you explore the full class dataset yourself, you’ll find that single exclusion wasn’t the whole story. This addendum walks through what’s actually in the data, and what it means for the module’s own outlier framework.

The headline numbers

  <div class="stat-chip chip-red">Mean: ${analysis.mean.toFixed(0)} ms</div>
  <div class="stat-chip chip-orange">Median: ${analysis.median.toFixed(0)} ms</div>
  <div class="stat-chip chip-red">SD: ${analysis.sd.toFixed(0)} ms</div>
</div>`

For genuine human reaction times, the mean should sit close to the median, and the standard deviation should be a fraction of the mean — not larger than it.

Here, the mean is × the median, and the standard deviation is larger than the mean itself. That gap is a sign the data is being pulled around by more than one extreme value.

Trials 1–2 account for % of flagged trials — well above the 20% you’d expect by chance. This matches exactly what we noticed live, watching one student’s data: the first trial or two, before you’ve settled into the task, is genuinely slower.

What explains the rest?

The trial-1–2 effect only accounts for part of the picture. The remaining flagged trials are scattered across trials 3–10, with no link to device or caffeine.

The likely explanation is the one already named live in the workshop: inattention — talking to a neighbour, looking away, getting distracted mid-task — just happening more often, across more of the room, than a single case.

This data can’t prove that — we didn’t measure distraction directly, so it’s an inference from the pattern, not something we captured directly.

Back to the framework

For every extreme value, this module already asks three questions. Applied at this scale, not just to one point:

Question Applied here
Is it an error? A little of it — the single ~56-second trial almost certainly is
Is it real, but not what we’re studying? Most of it — genuine slow taps, but driven by distraction, not the underlying reaction-time ability the business actually cares about
Is it a genuine feature of the process? Partly — the trial-1–2 warm-up effect is a real, relevant feature of how people perform this specific task

Different flagged trials plausibly belong in different rows of this table — a harder, more realistic version of the judgement call the workshop introduced.

What should we actually do with it?

Four decisions, applied consistently rather than case by case:

  • For this dataset, use the median and IQR, not the mean and SD, as the trustworthy summary — they held up through all of this; the mean and SD didn’t

Bridge to Week 2

We’ve just seen that a sample this noisy can produce a mean that’s × the median, purely from a messy minority of trials.

That’s exactly why you can’t just eyeball a mean, or a difference between two groups, and call it real. Next week: how we decide, systematically, which differences in a sample are worth taking seriously — and which are just noise.

Takeaways

  • A single “obvious” outlier is the easy case. Real data is often messier — check for that explicitly, don’t assume you’re done after removing one point
  • Always ask how much of the data is affected, and how it’s distributed across your groups (here: students, devices, trial position) — not just whether outliers exist
  • The three-question outlier framework still applies, but different points in the same dataset can land in different answers
  • Robust statistics — the median, the IQR — held up here exactly as intended. The mean and SD didn’t
